  #!/bin/bash

  set -u

  SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

  source "$SCRIPT_DIR/lib/reference-handler.sh"
  source "$SCRIPT_DIR/lib/options-parser.sh"
  source "$SCRIPT_DIR/lib/error-handler.sh"

  main() {
    local -A opts=(
      [dry_run]=false
      [verbose]=false
    )

    parse_options opts "$@"

    local source_file="$1"

    # 검증
    validate_path "$source_file" || handle_error 1 "Invalid path"

    # 파일 복사
    copy_markdown_file "$source_file" "${opts[dry_run]}"

    # 이미지 동기화

    # 참조 동기화
    sync_referenced_files "$source_file" "${opts[dry_run]}"
  }

  main "$@"
